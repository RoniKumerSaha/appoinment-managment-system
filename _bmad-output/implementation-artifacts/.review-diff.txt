+-- E0-S1: minimal seed-only schema.
+--
+-- This migration creates the tables the minimum seed (one clinic, one branch,
+-- one doctor, one receptionist, one admin, zero patients) needs to write to.
+-- E0-S2 widens this schema with appointment, audit, otp, session tables plus
+-- the booking concurrency primitives (EXCLUDE USING gist, SERIALIZABLE).
+--
+-- Conventions:
+--   - Primary keys: uuid, generated by the database (uuid_generate_v4 requires
+--     pgcrypto in production; for E0-S1 we use gen_random_uuid() which Postgres
+--     13+ ships in pgcrypto-equivalent without an explicit extension).
+--   - Timestamps: timestamptz, stored UTC.
+--   - `phone_number` is the natural identifier for patients — UNIQUE enforced
+--     here so two Patient rows for the same phone cannot exist (architecture §3.1).
+--   - `branch.tz` reserved for production time zone semantics; E0-S2 widens it.
+
+CREATE TABLE IF NOT EXISTS clinic (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  name         text NOT NULL,
+  created_at   timestamptz NOT NULL DEFAULT now()
+);
+
+CREATE TABLE IF NOT EXISTS branch (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  clinic_id    uuid NOT NULL REFERENCES clinic(id) ON DELETE RESTRICT,
+  name         text NOT NULL,
+  tz           text NOT NULL DEFAULT 'UTC',
+  created_at   timestamptz NOT NULL DEFAULT now()
+);
+
+CREATE TABLE IF NOT EXISTS doctor (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  full_name    text NOT NULL,
+  created_at   timestamptz NOT NULL DEFAULT now()
+);
+
+-- Many-to-many: doctors work at one or more branches.
+CREATE TABLE IF NOT EXISTS doctor_branch (
+  doctor_id    uuid NOT NULL REFERENCES doctor(id) ON DELETE CASCADE,
+  branch_id    uuid NOT NULL REFERENCES branch(id) ON DELETE CASCADE,
+  PRIMARY KEY (doctor_id, branch_id)
+);
+
+CREATE TABLE IF NOT EXISTS receptionist (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  branch_id    uuid NOT NULL REFERENCES branch(id) ON DELETE RESTRICT,
+  full_name    text NOT NULL,
+  created_at   timestamptz NOT NULL DEFAULT now()
+);
+
+CREATE TABLE IF NOT EXISTS admin_user (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  full_name    text NOT NULL,
+  created_at   timestamptz NOT NULL DEFAULT now()
+);
+
+-- Patient identified by phone_number — UNIQUE is the contract.
+CREATE TABLE IF NOT EXISTS patient (
+  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  phone_number text NOT NULL UNIQUE,
+  full_name    text,
+  created_at   timestamptz NOT NULL DEFAULT now()
+);

+/**
+ * Minimum seed: 1 clinic, 1 branch, 1 doctor + 1 doctor_branch row,
+ * 1 receptionist, 1 admin, 0 patients.
+ *
+ * Idempotency: every write is ON CONFLICT DO NOTHING keyed by the unique
+ * identifiers (phone_number for patients, name+branch_id for receptionist,
+ * etc.). Running this twice produces identical row counts; the seed test
+ * asserts equality.
+ */
+import { getPool, closePool } from '../../src/db/pool.js';
+
+const CLINIC_NAME = 'Test Clinic';
+const BRANCH_NAME = 'Branch 1';
+const BRANCH_TZ = 'UTC';
+const DOCTOR_NAME = 'Dr. Test';
+const RECEPTIONIST_NAME = 'Receptionist Test';
+const ADMIN_NAME = 'Admin Test';
+
+async function seed(): Promise<void> {
+  const pool = getPool();
+  const client = await pool.connect();
+  try {
+    await client.query('BEGIN');
+
+    // Clinic + branch (idempotent by name).
+    const clinicRes = await client.query<{ id: string }>(
+      `INSERT INTO clinic (name) VALUES ($1)
+       ON CONFLICT DO NOTHING
+       RETURNING id`,
+      [CLINIC_NAME],
+    );
+    let clinicId: string;
+    if (clinicRes.rows[0]) {
+      clinicId = clinicRes.rows[0].id;
+    } else {
+      const found = await client.query<{ id: string }>(
+        `SELECT id FROM clinic WHERE name = $1`,
+        [CLINIC_NAME],
+      );
+      const row = found.rows[0];
+      if (!row) throw new Error('seed_clinic_lookup_failed');
+      clinicId = row.id;
+    }
+
+    const branchRes = await client.query<{ id: string }>(
+      `INSERT INTO branch (clinic_id, name, tz) VALUES ($1, $2, $3)
+       ON CONFLICT DO NOTHING
+       RETURNING id`,
+      [clinicId, BRANCH_NAME, BRANCH_TZ],
+    );
+    let branchId: string;
+    if (branchRes.rows[0]) {
+      branchId = branchRes.rows[0].id;
+    } else {
+      const found = await client.query<{ id: string }>(
+        `SELECT id FROM branch WHERE clinic_id = $1 AND name = $2`,
+        [clinicId, BRANCH_NAME],
+      );
+      const row = found.rows[0];
+      if (!row) throw new Error('seed_branch_lookup_failed');
+      branchId = row.id;
+    }
+
+    // Doctor + doctor_branch.
+    const doctorRes = await client.query<{ id: string }>(
+      `INSERT INTO doctor (full_name) VALUES ($1)
+       ON CONFLICT DO NOTHING
+       RETURNING id`,
+      [DOCTOR_NAME],
+    );
+    let doctorId: string;
+    if (doctorRes.rows[0]) {
+      doctorId = doctorRes.rows[0].id;
+    } else {
+      const found = await client.query<{ id: string }>(
+        `SELECT id FROM doctor WHERE full_name = $1`,
+        [DOCTOR_NAME],
+      );
+      const row = found.rows[0];
+      if (!row) throw new Error('seed_doctor_lookup_failed');
+      doctorId = row.id;
+    }
+
+    await client.query(
+      `INSERT INTO doctor_branch (doctor_id, branch_id) VALUES ($1, $2)
+       ON CONFLICT DO NOTHING`,
+      [doctorId, branchId],
+    );
+
+    // Receptionist (one per branch — E0-S5 widens to two branches).
+    await client.query(
+      `INSERT INTO receptionist (branch_id, full_name)
+       SELECT $1, $2
+       WHERE NOT EXISTS (
+         SELECT 1 FROM receptionist WHERE branch_id = $1 AND full_name = $2
+       )`,
+      [branchId, RECEPTIONIST_NAME],
+    );
+
+    // Admin.
+    await client.query(
+      `INSERT INTO admin_user (full_name)
+       SELECT $1
+       WHERE NOT EXISTS (
+         SELECT 1 FROM admin_user WHERE full_name = $1
+       )`,
+      [ADMIN_NAME],
+    );
+
+    await client.query('COMMIT');
+    // eslint-disable-next-line no-console
+    console.log('seed_complete');
+  } catch (err) {
+    await client.query('ROLLBACK');
+    throw err;
+  } finally {
+    client.release();
+  }
+}
+
+const invokedDirectly =
+  process.argv[1] && process.argv[1].endsWith('01_minimum.ts');
+
+if (invokedDirectly) {
+  seed()
+    .catch((err) => {
+      // eslint-disable-next-line no-console
+      console.error('seed_failed', err);
+      process.exitCode = 1;
+    })
+    .finally(() => closePool());
+}
+
+export { seed };

+/**
+ * Express app factory (no side effects on import).
+ *
+ * `server.ts` is the runnable entry that binds to PORT. Tests import this
+ * module directly so importing does NOT start listening on a real port.
+ */

+import { healthRouter } from './routes/health.js';
+import { metricsRouter } from './routes/metrics.js';
+import { authMiddleware } from './middleware/auth.js';
+import { initLogContext, logger, requestLogger } from './middleware/log.js';
+
+export function createApp(): express.Express {
+  const app = express();
+  app.disable('x-powered-by');
+  app.use(express.json({ limit: '64kb' }));
+  app.use(requestLogger);
+  app.use(initLogContext);
+
+  app.use(healthRouter);
+  app.use(metricsRouter);
+
+  app.use(authMiddleware);
+
+  app.use(
+    (
+      err: unknown,
+      _req: express.Request,
+      res: express.Response,
+      // eslint-disable-next-line @typescript-eslint/no-unused-vars
+      _next: express.NextFunction,
+    ) => {

+      logger.error({ err: message }, 'request_failed');
+      res.status(500).json({ error: 'internal_error' });
+    },
+  );
+
+  return app;

+/**
+ * pg.Pool wrapper.
+ *
+ * - Reads DATABASE_URL from env (raw-SQL mode; no ORM).
+ * - Pool is created lazily so the process can start before Postgres is reachable.
+ * - `pingDb()` is what /readyz calls; failures surface as a typed error rather
+ *   than a thrown stack trace.
+ */
+import pg from 'pg';
+
+const { Pool } = pg;
+
+let pool: pg.Pool | null = null;
+
+export function getPool(): pg.Pool {
+  if (!pool) {
+    const connectionString = process.env.DATABASE_URL;
+    if (!connectionString) {
+      throw new Error('DATABASE_URL is not set');
+    }
+    pool = new Pool({
+      connectionString,
+      // Keep retry behaviour obvious: pool's underlying connect() retries per pg defaults.
+      max: 10,
+      idleTimeoutMillis: 30_000,
+      connectionTimeoutMillis: 2_000,
+    });
+    pool.on('error', (err) => {
+      // Idle client errors must not crash the process.
+      // Surface via the structured logger when wired; for now, last-resort log.
+      // eslint-disable-next-line no-console
+      console.error('[pg.Pool] idle client error', err.message);
+    });
+  }
+  return pool;
+}
+
+export async function pingDb(): Promise<boolean> {
+  try {
+    const client = await getPool().connect();
+    try {
+      await client.query('SELECT 1');
+      return true;
+    } finally {
+      client.release();
+    }
+  } catch {
+    return false;
+  }
+}
+
+export async function closePool(): Promise<void> {
+  if (pool) {
+    await pool.end();
+    pool = null;
+  }
+}

+/**

+  'booking.attempt',
+  'booking.success',
+  'booking.conflict',
+  'booking.failure',
+  'otp.verification',
+  'recurring.created',
+  'recurring.skipped',
+] as const;

+export type CounterName = (typeof NAMED_COUNTERS)[number];
+
+const state: Record<CounterName, number> = NAMED_COUNTERS.reduce(
+  (acc, name) => {
+    acc[name] = 0;
+    return acc;
+  },

+);

+  state[name] = (state[name] ?? 0) + by;

+}

+/**
+ * Auth middleware (STUB).
+ *
+ * E0-S3 replaces this with a real provider (JWT, opaque cookie, etc.). For now
+ * we accept the envelope shape from a header (`x-auth-claim`) ONLY so test code
+ * and tooling can exercise claim-aware guards. The token (when present) must be
+ * a JSON string of `AuthClaim`; an invalid shape yields anonymous context.
+ *
+ * Hard rules established by this stub:
+ *   - Body fields named role/branch_ids/scope/doctor_id are NEVER consulted.
+ *   - Staff tokens with empty branch_ids are rejected at middleware.
+ *   - Doctor tokens missing doctor_id are accepted but flagged for downstream
+ *     doctor-only guards (which reject at the route, not here).
+ */
+import type { Request, Response, NextFunction } from 'express';
+import type { AuthClaim, AuthContext } from '../types/auth.js';
+import { attachLogContext } from './log.js';
+
+function isRole(v: unknown): v is AuthClaim['role'] {
+  return v === 'patient' || v === 'receptionist' || v === 'doctor' || v === 'admin';
+}
+
+function isScope(v: unknown): v is AuthClaim['scope'] {
+  return v === 'self' || v === 'branch' || v === 'clinic' || v === 'global';
+}
+
+function parseClaimHeader(raw: string | undefined): AuthClaim | null {
+  if (!raw) return null;
+  let parsed: unknown;
+  try {
+    parsed = JSON.parse(raw);
+  } catch {
+    return null;
+  }
+  if (!parsed || typeof parsed !== 'object') return null;
+  const c = parsed as Record<string, unknown>;
+  if (typeof c.sub !== 'string') return null;
+  if (!isRole(c.role)) return null;
+  if (!Array.isArray(c.branch_ids)) return null;
+  if (!c.branch_ids.every((id) => typeof id === 'string')) return null;
+  if (!isScope(c.scope)) return null;
+  if (c.doctor_id !== undefined && typeof c.doctor_id !== 'string') return null;
+  const claim: AuthClaim = {
+    sub: c.sub,
+    role: c.role,
+    branch_ids: c.branch_ids,
+    scope: c.scope,
+    ...(typeof c.doctor_id === 'string' ? { doctor_id: c.doctor_id } : {}),
+  };
+  return claim;
+}
+
+declare module 'express' {
+  interface Request {
+    auth?: AuthContext;
+  }
+}
+
+/**
+ * Auth middleware. Always populates `req.auth` (anonymous when no/bad header).
+ * Staff with empty branch_ids is rejected at 403 — invariant from spec §Auth.
+ */
+export function authMiddleware(
+  req: Request,
+  res: Response,
+  next: NextFunction,
+): void {
+  const claim = parseClaimHeader(req.header('x-auth-claim'));
+
+  if (claim && claim.role !== 'patient' && claim.branch_ids.length === 0) {
+    res.status(403).json({ error: 'staff_token_requires_branch_ids' });
+    return;
+  }
+
+  req.auth = { claim };
+  if (claim) {
+    attachLogContext(req, {
+      actor_role: claim.role,
+      actor_id: claim.sub,
+    });
+  }
+  next();
+}

+/**
+ * Structured-log middleware.
+ *
+ * Contract fields every log line carries:
+ *   request_id, actor_role, actor_id, appointment_id (when in scope)
+ *
+ * Redaction (exact-key): password, phone_number, code. These keys are removed
+ * before pino serializes the log line; a value containing them in a deeper
+ * path is a reviewer-blocker because pino redact only walks the exact paths
+ * listed. We list the dotted paths the API surface actually emits today.
+ *
+ * The log middleware also exposes a tiny test-only endpoint behaviour: any
+ * handler can call `req.log.info({ appointment_id, ... }, 'msg')` and the four
+ * fields will be present in the JSON line.
+ */
+import type { Request, Response, NextFunction } from 'express';
+import pino from 'pino';
+import pinoHttp from 'pino-http';
+
+const redactPaths = [
+  'req.headers.authorization',
+  'req.headers.cookie',
+  'req.body.password',
+  'req.body.phone_number',
+  'req.body.code',
+  'res.headers["set-cookie"]',
+];
+
+const baseLogger = pino({
+  level: process.env.LOG_LEVEL ?? 'info',
+  redact: { paths: redactPaths, censor: '[REDACTED]' },
+  base: { service: 'clinic-api' },
+  timestamp: pino.stdTimeFunctions.isoTime,
+});
+
+export const logger = baseLogger;
+
+export const requestLogger = pinoHttp({
+  logger: baseLogger,
+  customLogLevel: (_req, res, err) => {
+    if (err || res.statusCode >= 500) return 'error';
+    if (res.statusCode >= 400) return 'warn';
+    return 'info';
+  },
+  // Attach request_id on every line so structured-log contract holds even
+  // before any handler logs explicitly.
+  genReqId: (req, res) => {
+    const incoming = (req.headers['x-request-id'] as string | undefined) ?? '';
+    const id = incoming || cryptoRandom();
+    res.setHeader('x-request-id', id);
+    return id;
+  },
+  customProps: (req) => {
+    // actor_role / actor_id / appointment_id populate when the handler sets them
+    // on the request via attachLogContext(). The fields are always present in
+    // the JSON envelope (null when unset) so log consumers can rely on shape.
+    const ctx = (req as Request & { logCtx?: LogContext }).logCtx ?? {};
+    return {
+      actor_role: ctx.actor_role ?? null,
+      actor_id: ctx.actor_id ?? null,
+      appointment_id: ctx.appointment_id ?? null,
+    };
+  },
+});
+
+export interface LogContext {
+  actor_role?: string | null;
+  actor_id?: string | null;
+  appointment_id?: string | null;
+}
+
+/**
+ * Attach mutable log context to a request so subsequent logs in the same
+ * request carry the same actor_role / actor_id / appointment_id values.
+ * Auth middleware calls this once the claim is parsed.
+ */
+export function attachLogContext(
+  req: Request,
+  ctx: Partial<LogContext>,
+): void {
+  const r = req as Request & { logCtx?: LogContext };
+  r.logCtx = { ...(r.logCtx ?? {}), ...ctx };
+}
+
+function cryptoRandom(): string {
+  // Node 20 has globalThis.crypto.randomUUID.
+  return globalThis.crypto.randomUUID();
+}
+
+/**
+ * Convenience middleware that just ensures the log context is initialised so
+ * downstream handlers can mutate it safely.
+ */
+export function initLogContext(
+  _req: Request,
+  _res: Response,
+  next: NextFunction,
+): void {
+  next();
+}
