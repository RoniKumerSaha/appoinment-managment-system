+CREATE EXTENSION IF NOT EXISTS btree_gist;

+-- E0-S2: 0003 — shared functions and triggers.
+--
+-- app_now(): returns current timestamptz in UTC. We use `SELECT now()` directly
+-- because `now()` already returns timestamptz in UTC (Postgres stores it as
+-- UTC internally and renders it in the session's timezone). The earlier draft
+-- `now() AT TIME ZONE 'UTC'` was wrong: that expression returns
+-- `timestamp without time zone`, and storing it back in a `timestamptz`
+-- column would re-interpret the value as session-local time.
+--
+-- STABLE (not IMMUTABLE) because `now()` is volatile within a transaction —
+-- which is exactly what we want for column defaults (statement-time value).
+--
+-- set_updated_at(): BEFORE UPDATE trigger function. Sets NEW.updated_at to
+-- app_now(). Returns NEW explicitly so the UPDATE is not silently dropped.
+
+CREATE OR REPLACE FUNCTION app_now() RETURNS timestamptz
+  LANGUAGE sql
+  STABLE
+AS $$
+  SELECT now();
+$$;
+
+CREATE OR REPLACE FUNCTION set_updated_at() RETURNS trigger
+  LANGUAGE plpgsql
+AS $$
+BEGIN
+  NEW.updated_at := app_now();
+  RETURN NEW;
+END;
+$$;

+-- E0-S2: 0008 — appointment.
+--
+-- The booking spine. Carries Architecture §4 invariants:
+--   - status enum is exactly {Scheduled, Completed, Cancelled, No-show}
+--   - GIST exclusion prevents a doctor from having two Scheduled appointments
+--     at the same (branch, time window). Half-open ranges '[)' so back-to-back
+--     appointments like [10:00, 10:30) and [10:30, 11:00) do NOT collide.
+--   - Partial unique (patient_phone, doctor_id) WHERE status='Scheduled'
+--     prevents a patient from having two Scheduled appointments with the SAME
+--     doctor. Cross-doctor Scheduled appointments ARE allowed (Architecture
+--     §4 invariant 3 is per-doctor; honors FR-028 "patient may have future
+--     appointments with multiple doctors").
+--   - Two read-path B-tree indexes for availability latency (Architecture §6):
+--     (patient_phone, status, start_datetime) and
+--     (doctor_id, branch_id, start_datetime).
+--
+-- FKs use ON DELETE RESTRICT — appointment rows are never deleted (Cancelled is
+-- the terminal state), but a parent DELETE would orphan the rows; RESTRICT
+-- fails loud instead of cascading silently.
+
+-- ===== enum =====
+DO $$
+BEGIN
+  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'appointment_status') THEN
+    CREATE TYPE appointment_status AS ENUM (
+      'Scheduled',
+      'Completed',
+      'Cancelled',
+      'No-show'
+    );
+  END IF;
+END $$;
+
+-- ===== table =====
+CREATE TABLE IF NOT EXISTS appointment (
+  id                  uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  clinic_id           uuid NOT NULL REFERENCES clinic(id) ON DELETE RESTRICT,
+  branch_id           uuid NOT NULL REFERENCES branch(id) ON DELETE RESTRICT,
+  doctor_id           uuid NOT NULL REFERENCES doctor(id) ON DELETE RESTRICT,
+  patient_phone       text NOT NULL,
+  appointment_type_id uuid REFERENCES appointment_type(id) ON DELETE RESTRICT,
+  start_datetime      timestamptz NOT NULL,
+  end_datetime        timestamptz NOT NULL CHECK (end_datetime > start_datetime),
+  status              appointment_status NOT NULL DEFAULT 'Scheduled',
+  created_by          text NOT NULL CHECK (created_by ~ '^(staff|patient):[A-Za-z0-9+_.-]{1,255}$' AND length(created_by) <= 256),
+  created_at          timestamptz NOT NULL DEFAULT app_now(),
+  updated_at          timestamptz NOT NULL DEFAULT app_now()
+);

+-- E0-S2: 0009 — appointment_audit.
+--
+-- Append-only audit trail. Records every event (created, rescheduled,
+-- cancelled, completed, marked no-show, etc.) with the actor who performed it
+-- and a JSONB payload of event-specific data.
+--
+-- Append-only enforcement is the trigger `reject_audit_mutation`, which fires
+-- BEFORE UPDATE OR DELETE and RAISEs. REVOKE alone is insufficient because
+-- superuser bypasses it; the trigger is the primary guard.
+--
+-- appointment_id is FK NULLABLE because some events (e.g. attempted booking
+-- that conflicted) have no associated appointment row. The FK uses
+-- NO ACTION DEFERRABLE INITIALLY DEFERRED so that within a transaction that
+-- creates an appointment + writes its audit row, the audit's appointment_id
+-- resolves correctly even if the appointment row was inserted in the same
+-- transaction.
+
+CREATE TABLE IF NOT EXISTS appointment_audit (
+  id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
+  appointment_id uuid,
+  actor_role     text NOT NULL,
+  actor_id       text NOT NULL,
+  event          text NOT NULL,
+  payload        jsonb,
+  created_at     timestamptz NOT NULL DEFAULT app_now()
+);
+
+-- FK as a separate ALTER so DEFERRABLE works (DEFERRABLE on inline REFERENCES
+-- is fine in modern Postgres but explicit ALTER is clearer).
+DO $$
+BEGIN
+  IF NOT EXISTS (
+    SELECT 1 FROM pg_constraint WHERE conname = 'appointment_audit_appointment_id_fkey'
+  ) THEN
+    ALTER TABLE appointment_audit
+      ADD CONSTRAINT appointment_audit_appointment_id_fkey
+      FOREIGN KEY (appointment_id)
+      REFERENCES appointment(id)
+      ON DELETE NO ACTION
+      DEFERRABLE INITIALLY DEFERRED;
+  END IF;
+END $$;
+
+-- ===== append-only enforcement =====
+CREATE OR REPLACE FUNCTION reject_audit_mutation() RETURNS trigger
+  LANGUAGE plpgsql
+AS $$
+BEGIN
+  RAISE EXCEPTION 'appointment_audit_is_append_only';
+END;
+$$;
+
+DROP TRIGGER IF EXISTS trg_appointment_audit_no_update ON appointment_audit;
+CREATE TRIGGER trg_appointment_audit_no_update
+  BEFORE UPDATE ON appointment_audit
+  FOR EACH ROW EXECUTE FUNCTION reject_audit_mutation();
+
+DROP TRIGGER IF EXISTS trg_appointment_audit_no_delete ON appointment_audit;
+CREATE TRIGGER trg_appointment_audit_no_delete
+  BEFORE DELETE ON appointment_audit
+  FOR EACH ROW EXECUTE FUNCTION reject_audit_mutation();

+CREATE INDEX IF NOT EXISTS idx_appointment_audit_actor
+  ON appointment_audit(actor_role, actor_id);
+CREATE INDEX IF NOT EXISTS idx_appointment_audit_event_created
+  ON appointment_audit(event, created_at DESC);

+/**
+ * DB-touching test helper.
+ *
+ * Two operating modes:
+ *
+ * 1. **TESTCONTAINERS mode (CI default):** spins up a `postgres:16-alpine`
+ *    container, applies all migrations, yields a pool. The container is stopped
+ *    on every callback exit (normal or thrown). Used when `DATABASE_URL` is not
+ *    set and Testcontainers is available.
+ *
+ * 2. **DATABASE_URL mode (local dev):** uses an existing `DATABASE_URL` (the
+ *    migrations must already be applied). No container lifecycle to manage.
+ *    Selected automatically when `DATABASE_URL` is present in the env.
+ *
+ * The test container is created once per worker via `globalSetup.ts` and the
+ * resulting pool is reused across `migrations.test.ts`, `invariants.test.ts`,
+ * and `seed.test.ts` — re-booting per file would double migration time and
+ * risk clock-skew between containers.
+ */
+import pg from 'pg';
+import { readFileSync, readdirSync } from 'node:fs';
+import { dirname, resolve, join } from 'node:path';
+import { fileURLToPath } from 'node:url';
+
+const { Pool } = pg;
+const here = dirname(fileURLToPath(import.meta.url));
+const migrationsDir = resolve(here, '..', '..', 'db', 'migrations');
+
+export interface TestDb {
+  pool: pg.Pool;
+  close: () => Promise<void>;
+}
+
+let sharedDb: TestDb | null = null;
+
+async function applyMigrations(pool: pg.Pool): Promise<void> {
+  // Forward-only; ignore already-applied (we boot a fresh DB each time).
+  const files = readdirSync(migrationsDir)
+    .filter((f) => /^\d{4}_.*\.sql$/.test(f) && !f.includes('.down.'))
+    .sort();
+  for (const file of files) {
+    const sql = readFileSync(join(migrationsDir, file), 'utf8');
+    await pool.query(sql);
+  }
+}
+
+async function fromEnv(): Promise<TestDb> {
+  const url = process.env['DATABASE_URL'];
+  if (!url) throw new Error('DATABASE_URL not set and Testcontainers not enabled');
+  const pool = new Pool({ connectionString: url });
+  return {
+    pool,
+    close: async () => {
+      await pool.end();
+    },
+  };
+}
+
+async function fromTestcontainers(): Promise<TestDb> {
+  // Lazy import so dev environments without testcontainers don't fail at parse time.
+  // eslint-disable-next-line @typescript-eslint/no-var-requires
+  const { PostgreSqlContainer } = await import('@testcontainers/postgresql');
+  const container = await new PostgreSqlContainer('postgres:16-alpine').start();
+  const pool = new Pool({ connectionString: container.getConnectionUri() });
+  await applyMigrations(pool);
+  return {
+    pool,
+    close: async () => {
+      await pool.end();
+      await container.stop();
+    },
+  };
+}
+
+/**
+ * Get (or create) the shared test DB. Subsequent calls return the same pool.
+ * Always call `closeShared()` in suite teardown (or rely on process exit).
+ */
+export async function getTestDb(): Promise<TestDb> {
+  if (sharedDb) return sharedDb;
+  sharedDb = process.env['DATABASE_URL']
+    ? await fromEnv()
+    : await fromTestcontainers();
+  return sharedDb;
+}
+
+export async function closeShared(): Promise<void> {
+  if (sharedDb) {
+    await sharedDb.close();
+    sharedDb = null;
+  }
+}
+
+export async function withTestDb<T>(fn: (db: TestDb) => Promise<T>): Promise<T> {
+  const db = await getTestDb();
+  try {
+    return await fn(db);
+  } finally {
+    // Intentionally NOT closing the shared pool here — see globalSetup.ts for
+    // worker-level teardown. Per-call close would defeat the shared container.
+  }
+}
